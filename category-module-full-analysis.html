1. نظرة عامة على الوحدة

ما هي وحدة Category؟

وحدة Category هي المسؤولة عن إدارة التصنيفات في نظام ThreadMarket SaaS. التصنيفات تُستخدم لتنظيم المنتجات وتسهيل البحث والتصفح على المتجر.

الملفات الموجودة حالياً

  • category.model.ts — التعامل مع قاعدة البيانات
  • category.service.ts — طبقة الخدمات
  • category.controller.ts — التعامل مع الطلبات
  • category.routes.ts — تعريف مسارات API
  • category.validators.ts — ملف فارغ (يحتاج تطوير)
src/modules/category/

شرح المكونات

  1. Routes — يحدد endpoint و method (GET فقط حالياً)
  2. Middleware — authenticate يتحقق من JWT token، authorize يتحقق من الصلاحيات
  3. Controller — يستقبل الطلبات ويُجهز الاستجابة (يحتوي على todo comment)
  4. Service — طبقة وسيطة (حالياً مجرد pass-through بدون منطق)
  5. Model — التعامل المباشر مع Prisma ORM (دالة واحدة فقط)
  6. Validators — ملف فارغ بالكامل ❌

4. تحليل الملفات واحداً تلو الآخر

📄 category.model.ts — نموذج البيانات بنية تحسين

import prisma from "../../config/database"; class categoryModel { async getCategories(filter: { search: string; skip?: number; take?: number; }) { return await prisma.category.findMany({ where: { OR: [ { name: { search: filter.search } }, { description: { search: filter.search } }, ] } skip: filter.skip, take: filter.take, }) } } export const category = new categoryModel();

الحالة الحالية

  • دالة واحدة فقط: getCategories
  • يبحث في الاسم والوصف باستخدام search
  • يدعم الترقيم (skip/take)

المشاكل

  • بنية اسم الكلاس categoryModel — يجب أن يكون CategoryModel
  • بنية لا يوجد TypeScript interfaces
  • تحسين لا يوجد CRUD كامل
  • تحسين لا يوجد ترتيب (ordering)
  • تحسين لا يوجد تصفية حسب isHidden
  • تحسين لا يُرجع بيانات pagination (total, totalPages)
  • أمان لا يوجد try-catch لمعالجة الأخطاء

📄 category.service.ts — طبقة الخدمات بنية

import { category } from "./category.model"; class CategoryService { async getCategories(filter: { search: string; skip?: number; take?: number; }) { return await category.getCategories(filter); } } export const categoryService = new CategoryService();

المشاكل

  • الخدمة حالياً مجرد pass-through — لا يوجد منطق أعمال
  • لا يوجد CRUD كامل
  • لا يوجد معالجة أخطاء
  • لا يوجد تحويل بيانات أو تحقق

📄 category.controller.ts — متحكم الطلبات خطأ تحسين

export const getCategories = asyncHandler( async (req: Request, res: Response) => { // todo => Discuss about category ownership const { search, limit = 10, page = 1, } = req.params as { search: string; limit: string; page: string; }; const skip = (Number(page) - 1) * Number(limit); const filter = { search: search, skip: skip, take: Number(limit), }; const categories = await categoryService.getCategories(filter); if (!categories || categories.length === 0) { throw new AppError("Categories not found", StatusCodes.NOT_FOUND); } res.status(StatusCodes.OK).json({ success: true, message: "OK", data: { categories, total: categories.length, page: Number(page), }, }); }, );

المشاكل

  • خطأ يقرأ القيم من req.params بدلاً من req.query — المفترض query parameters للبحث
  • خطأ total: categories.length — هذا عدد النتائج المرتجعة وليس العدد الإجمالي!
  • تحسين لا يوجد CRUD كامل
  • تحسين لا يوجد todo comment لم يُحل
  • تحسين لا يوجد ترتيب أو فلترة حسب isHidden

📄 category.routes.ts — المسارات بنية

import { Router } from "express"; import authenticate from "../../shared/middleware/authenticate"; import { authorize } from "../../shared/middleware/authorize"; import { getCategories } from "./category.controller"; const router = Router(); router.use(authenticate); router.route("/").get(getCategories); export default router;

المشاكل

  • يوجد authorize مستورد لكن غير مستخدم فعلياً
  • لا يوجد routes للإنشاء أو التحديث أو الحذف
  • البحث يتم عبر params وليس query string

📄 category.validators.ts — المحققات خطأ

الملف فارغ بالكامل ❌ — لا يوجد أي تحقق من صحة المدخلات.

ما المطلوب؟

  • التحقق من صحة الـ query parameters (search, page, limit)
  • التحقق من بيانات الإنشاء (name, description, media)
  • التحقق من بيانات التحديث
  • التحقق من صحة الـ ID في الطلبات

7. الكود المحسن النهائي

📄 category.model.ts — النسخة المحسنة الكاملة

import prisma from "../../config/database"; import AppError from "../../shared/errors/AppError"; import { StatusCodes } from "http-status-codes"; import { Prisma } from "@prisma/client"; export interface CategoryFilter { search?: string; skip?: number; take?: number; sortBy?: string; sortOrder?: 'asc' | 'desc'; isHidden?: boolean; } export interface CategoryResult<T> { data: T[]; total: number; page: number; limit: number; totalPages: number; } class CategoryModel { private buildWhereClause(filter: CategoryFilter): Prisma.CategoryWhereInput { const where: Prisma.CategoryWhereInput = {}; if (filter.isHidden === undefined || filter.isHidden === false) { where.ishidden = false; } if (filter.search) { where.OR = [ { name: { search: filter.search } }, { description: { search: filter.search } }, ]; } return where; } async getCategories(filter: CategoryFilter): Promise<CategoryResult<any>> { try { const where = this.buildWhereClause(filter); const take = filter.take ?? 10; const skip = filter.skip ?? 0; const [total, data] = await Promise.all([ prisma.category.count({ where }), prisma.category.findMany({ where, skip, take, orderBy: filter.sortBy ? { [filter.sortBy]: filter.sortOrder ?? 'asc' } : { createdAt: 'desc' }, }), ]); return { data, total, page: Math.floor(skip / take) + 1, limit: take, totalPages: Math.ceil(total / take), }; } catch (error) { throw new AppError("فشل في جلب التصنيفات", StatusCodes.INTERNAL_SERVER_ERROR); } } async getCategoryById(id: string) { try { const category = await prisma.category.findUnique({ where: { id } }); if (!category) throw new AppError("التصنيف غير موجود", StatusCodes.NOT_FOUND); return category; } catch (error) { if (error instanceof AppError) throw error; throw new AppError("فشل في جلب التصنيف", StatusCodes.INTERNAL_SERVER_ERROR); } } async createCategory(data: Prisma.CategoryCreateInput) { try { const existing = await prisma.category.findUnique({ where: { name: data.name } }); if (existing) throw new AppError(`التصنيف "${data.name}" موجود مسبقاً`, StatusCodes.CONFLICT); return prisma.category.create({ data }); } catch (error) { if (error instanceof AppError) throw error; throw new AppError("فشل في إنشاء التصنيف", StatusCodes.INTERNAL_SERVER_ERROR); } } async updateCategory(id: string, data: Prisma.CategoryUpdateInput) { try { await this.getCategoryById(id); return prisma.category.update({ where: { id }, data }); } catch (error) { if (error instanceof AppError) throw error; throw new AppError("فشل في تحديث التصنيف", StatusCodes.INTERNAL_SERVER_ERROR); } } async deleteCategory(id: string, soft = true) { try { await this.getCategoryById(id); if (soft) return prisma.category.update({ where: { id }, data: { ishidden: true } }); return prisma.category.delete({ where: { id } }); } catch (error) { if (error instanceof AppError) throw error; throw new AppError("فشل في حذف التصنيف", StatusCodes.INTERNAL_SERVER_ERROR); } } async toggleVisibility(id: string) { const cat = await this.getCategoryById(id); return prisma.category.update({ where: { id }, data: { ishidden: !cat.ishidden } }); } } export const categoryModel = new CategoryModel();
