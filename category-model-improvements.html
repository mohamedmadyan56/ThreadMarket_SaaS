4. Case-Insensitive Search Bug

Problem

contains in Prisma is case-sensitive on PostgreSQL and MySQL with certain collations. Searching for "electronics" won't find "Electronics".

Suggested Fix

// Before where: { OR: [ { name: { contains: filter.search } }, { description: { contains: filter.search } }, ] } // After (case-insensitive) where: { OR: [ { name: { search: filter.search } }, { description: { search: filter.search } }, ] }

Note: search requires a @@@fulltext index in MySQL or uses ILIKE in PostgreSQL. Alternatively, use { mode: 'insensitive' } with contains if using PostgreSQL.

11. Improved Final Model Structure Structure

Complete Improved Model

import prisma from "../../config/database"; import AppError from "../../shared/errors/AppError"; import { StatusCodes } from "http-status-codes"; import { Prisma } from "@prisma/client"; export interface CategoryFilter { search?: string; skip?: number; take?: number; sortBy?: string; sortOrder?: 'asc' | 'desc'; isHidden?: boolean; } export interface CategoryResult<T> { data: T[]; total: number; page: number; limit: number; totalPages: number; } class CategoryModel { private buildWhereClause(filter: CategoryFilter): Prisma.CategoryWhereInput { const where: Prisma.CategoryWhereInput = {}; if (filter.isHidden === undefined || filter.isHidden === false) { where.ishidden = false; } if (filter.search) { where.OR = [ { name: { search: filter.search } }, { description: { search: filter.search } }, ]; } return where; } async getCategories(filter: CategoryFilter): Promise<CategoryResult<Prisma.CategoryGetPayload<{}>>> { try { const where = this.buildWhereClause(filter); const [total, data] = await Promise.all([ prisma.category.count({ where }), prisma.category.findMany({ where, skip: filter.skip ?? 0, take: filter.take ?? 10, orderBy: filter.sortBy ? { [filter.sortBy]: filter.sortOrder ?? 'asc' } : { createdAt: 'desc' }, }), ]); return { data, total, page: filter.skip !== undefined ? Math.floor(filter.skip / (filter.take ?? 10)) + 1 : 1, limit: filter.take ?? 10, totalPages: Math.ceil(total / (filter.take ?? 10)), }; } catch (error) { throw new AppError('Failed to fetch categories', StatusCodes.INTERNAL_SERVER_ERROR); } } async getCategoryById(id: string) { try { const category = await prisma.category.findUnique({ where: { id } }); if (!category) { throw new AppError('Category not found', StatusCodes.NOT_FOUND); } return category; } catch (error) { throw new AppError('Failed to fetch category', StatusCodes.INTERNAL_SERVER_ERROR); } } async createCategory(data: Prisma.CategoryCreateInput) { try { const existing = await prisma.category.findUnique({ where: { name: data.name } }); if (existing) { throw new AppError(`Category "${data.name}" already exists`, StatusCodes.CONFLICT); } return prisma.category.create({ data }); } catch (error) { throw new AppError('Failed to create category', StatusCodes.INTERNAL_SERVER_ERROR); } } async updateCategory(id: string, data: Prisma.CategoryUpdateInput) { try { await this.getCategoryById(id); return prisma.category.update({ where: { id }, data }); } catch (error) { throw new AppError('Failed to update category', StatusCodes.INTERNAL_SERVER_ERROR); } } async deleteCategory(id: string, soft = true) { try { if (soft) { return prisma.category.update({ where: { id }, data: { ishidden: true } }); } return prisma.category.delete({ where: { id } }); } catch (error) { throw new AppError('Failed to delete category', StatusCodes.INTERNAL_SERVER_ERROR); } } } export const categoryModel = new CategoryModel();
