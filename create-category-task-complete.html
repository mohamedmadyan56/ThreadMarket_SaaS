2. المشكلة: الـ Schema الحالي لا يدعم الـ Task

الحالة الحالية في Prisma Schema

model Category { id String @id @default(uuid()) @db.Uuid name String @unique @map("category_name") // فريد عالمياً! description String media Json ishidden Boolean @default(false) Products Product[] createdAt DateTime @default(now()) updateAt DateTime @default(now()) } // لا يوجد brandId!

المشاكل

  • لا يوجد brandId — Category غير مرتبة بأي Brand
  • name فريد عالمياً (@unique) — الـ Task مطلوب فريد داخل البراند فقط
  • الـ Task مطلوب imageUrl لكن الـ Schema فيه media (Json)

3. الخطوة 1: تعديل Prisma Schema

التعديل المطلوب في prisma/schema.prisma

model Category { id String @id @default(uuid()) @db.Uuid name String @map("category_name") description String @default("") imageUrl String? @map("image_url") ishidden Boolean @default(false) brandId String @map("brand_id") @db.Uuid brand Brand @relation(fields: [brandId], references: [id], onDelete: Cascade) Products Product[] createdAt DateTime @default(now()) updateAt DateTime @default(now()) @@unique([brandId, name]) @@map("categories") }

5. الخطوة 3: category.model.ts

category.model.ts src/modules/category/category.model.ts

الكود الكامل

import prisma from "../../config/database"; import AppError from "../../shared/errors/AppError"; import { StatusCodes } from "http-status-codes"; // ─── Interfaces ─── export interface CreateCategoryInput { name: string; imageUrl?: string; } // ─── Model ─── class CategoryModel { async createCategory(brandId: string, data: CreateCategoryInput) { try { // تحقق من عدم تكرار الاسم داخل نفس البراند const existing = await prisma.category.findFirst({ where: { brandId, name: data.name, }, }); if (existing) { throw new AppError( `Category "${data.name}" already exists for this brand`, StatusCodes.CONFLICT ); } return await prisma.category.create({ data: { name: data.name, imageUrl: data.imageUrl ?? null, brandId, }, }); } catch (error) { if (error instanceof AppError) throw error; throw new AppError( "Failed to create category", StatusCodes.INTERNAL_SERVER_ERROR ); } } async getCategoryById(id: string) { try { const category = await prisma.category.findUnique({ where: { id } }); if (!category) { throw new AppError("Category not found", StatusCodes.NOT_FOUND); } return category; } catch (error) { if (error instanceof AppError) throw error; throw new AppError( "Failed to fetch category", StatusCodes.INTERNAL_SERVER_ERROR ); } } async getCategoriesByBrand(brandId: string) { try { return await prisma.category.findMany({ where: { brandId }, orderBy: { createdAt: "desc" }, }); } catch (error) { throw new AppError( "Failed to fetch categories", StatusCodes.INTERNAL_SERVER_ERROR ); } } } export const categoryModel = new CategoryModel();

شرح الكود:

  • createCategory — يستقبل brandId و data، يتحقق من عدم التكرار داخل البراند، ثم ينشئ التصنيف
  • findFirst — يبحث عن تصنيف بنفس الاسم لنفس البراند (يتحقق من عدم التكرار)
  • getCategoryById — لجلب تصنيف واحد بالـ ID
  • getCategoriesByBrand — لجلب كل التصنيفات الخاصة ببراند معين

6. الخطوة 4: category.validators.ts

category.validators.ts src/modules/category/category.validators.ts

الكود الكامل

import { z } from "zod"; // ─── Schema: إنشاء تصنيف (POST /brands/:brandId/categories) ─── export const createCategorySchema = z.object({ brandId: z.string().uuid("Invalid brand ID format"), name: z .string() .min(1, "Category name is required") .max(100, "Category name must be at most 100 characters") .trim(), imageUrl: z.string().url("Invalid image URL").optional(), }); // ─── Schema: جلب تصنيفات البراند (GET /brands/:brandId/categories) ─── export const getCategoriesByBrandSchema = z.object({ brandId: z.string().uuid("Invalid brand ID format"), }); // ─── Schema: جلب تصنيف بالـ ID (GET /categories/:id) ─── export const getCategoryByIdSchema = z.object({ id: z.string().uuid("Invalid category ID format"), });

شرح:

  • brandId — من الـ params، يجب أن يكون UUID صحيح
  • name — مطلوب، 1-100 حرف، trimmed
  • imageUrl — اختياري،但如果 وُجد يجب أن يكون URL صحيح

8. الخطوة 6: category.controller.ts

category.controller.ts src/modules/category/category.controller.ts

الكود الكامل

import { asyncHandler } from "../../shared/utils/asyncHandler"; import { Request, Response } from "express"; import { StatusCodes } from "http-status-codes"; import { categoryService } from "./category.service"; // ─── POST /brands/:brandId/categories ─── export const createCategory = asyncHandler( async (req: Request, res: Response) => { const { brandId } = req.params; const { name, imageUrl } = req.body; const category = await categoryService.createCategory(brandId, { name, imageUrl, }); res.status(StatusCodes.CREATED).json({ success: true, message: "Category created successfully", data: { id: category.id, }, }); }, ); // ─── GET /brands/:brandId/categories ─── export const getCategoriesByBrand = asyncHandler( async (req: Request, res: Response) => { const { brandId } = req.params; const categories = await categoryService.getCategoriesByBrand(brandId); res.status(StatusCodes.OK).json({ success: true, message: "OK", data: { categories, }, }); }, ); // ─── GET /categories/:id ─── export const getCategoryById = asyncHandler( async (req: Request, res: Response) => { const { id } = req.params; const category = await categoryService.getCategoryById(id); res.status(StatusCodes.OK).json({ success: true, message: "OK", data: { category }, }); }, );

شرح:

  • brandId — من req.params (الـ URL)
  • name, imageUrl — من req.body
  • Response 201 — يرجع فقط الـ id كما هو مطلوب في الـ Task

9. الخطوة 7: category.routes.ts

category.routes.ts src/modules/category/category.routes.ts

الكود الكامل

import { Router } from "express"; import authenticate from "../../shared/middleware/authenticate"; import { authorize } from "../../shared/middleware/authorize"; import { validate } from "../../shared/middleware/validate"; import { createCategory, getCategoriesByBrand, getCategoryById, } from "./category.controller"; import { createCategorySchema, getCategoriesByBrandSchema, getCategoryByIdSchema, } from "./category.validators"; const router = Router(); // ─── All routes require authentication ─── router.use(authenticate); // ─── POST /brands/:brandId/categories ─── router.post( "/brands/:brandId/categories", authorize("brand"), validate(createCategorySchema), createCategory ); // ─── GET /brands/:brandId/categories ─── router.get( "/brands/:brandId/categories", authorize("brand"), validate(getCategoriesByBrandSchema), getCategoriesByBrand ); // ─── GET /categories/:id ─── router.get( "/categories/:id", validate(getCategoryByIdSchema), getCategoryById ); export default router;

شرح:

  • authorize("brand") — فقط المستخدمين برول brand يقدروا ينشئوا/يطلعوا تصنيفات
  • validate — يتحقق من صحة البيانات قبل الوصول للـ controller
  • router.use(authenticate) — كل الـ routes تحتاج JWT token
